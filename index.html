import "jsr:@supabase/functions-js/edge-runtime.d.ts"; const corsHeaders = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type", "Access-Control-Allow-Methods": "POST, OPTIONS", }; const DRD3M_API = "https://drd3m.me/api/v2"; function response( data: unknown, status = 200 ) { return new Response( JSON.stringify(data), { status, headers: { ...corsHeaders, "Content-Type": "application/json", }, } ); } async function drd3m( params: Record ) { const key = Deno.env.get("DRD3M_API_KEY"); if(!key){ throw new Error( "DRD3M_API_KEY is missing" ); } const form = new URLSearchParams(); form.append("key", key); for( const [name,value] of Object.entries(params) ){ form.append(name,value); } const result = await fetch( DRD3M_API, { method:"POST", headers:{ "Content-Type": "application/x-www-form-urlencoded" }, body:form } ); const text = await result.text(); let data; try{ data = JSON.parse(text); }catch{ data = { raw:text }; } if(!result.ok){ throw new Error( "DrD3M returned HTTP " + result.status ); } return data; } Deno.serve( async(req)=>{ if(req.method === "OPTIONS"){ return new Response( "ok", { headers: corsHeaders } ); } if(req.method !== "POST"){ return response( { success:false, error:"POST only" }, 405 ); } try{ const body = await req.json(); const action = body.action; if(action === "services"){ const data = await drd3m({ action:"services" }); const markup = 1 + Number( Deno.env.get( "MARKUP_PERCENT" ) || "30" ) / 100; if(Array.isArray(data)){ const services = data.map( (service:any)=>{ const rate = Number( service.rate || 0 ); return { service: service.service, name: service.name, type: service.type, rate: rate, base_rate: rate, sell_rate: Number( ( rate * markup ).toFixed(6) ), min: Number( service.min || 0 ), max: Number( service.max || 0 ), dripfeed: service.dripfeed, refill: service.refill, cancel: service.cancel, category: service.category }; } ); return response({ success:true, services }); } return response({ success:true, services:data }); } if(action === "status"){ const order = String( body.order || "" ); if(!order){ return response({ success:false, error: "Order ID required" },400); } const data = await drd3m({ action:"status", order }); return response({ success:true, status:data }); } if(action === "balance"){ const data = await drd3m({ action:"balance" }); return response({ success:true, balance:data }); } if(action === "create_order"){ const service = String( body.service || "" ); const link = String( body.link || "" ); const quantity = Number( body.quantity || 0 ); if( !service || !link || !quantity ){ return response({ success:false, error: "Missing order data" },400); } if(quantity <= 0){ return response({ success:false, error: "Invalid quantity" },400); } const data = await drd3m({ action:"add", service, link, quantity: String(quantity) }); return response({ success:true, order:data }); } return response({ success:false, error: "Unknown action" },400); }catch(error){ console.error(error); return response({ success:false, error: error instanceof Error ? error.message : "Server error" },500); } });
